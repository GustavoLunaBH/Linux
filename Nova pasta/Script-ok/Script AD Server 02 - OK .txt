#!/bin/bash
# adserver02_complete.sh
# Script para configuração do Samba AD como Controlador Secundário
# Versão: 1.4 - Corrigido comando join com sintaxe correta

set -e  # Para o script em caso de erro

# Cores para output
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m' # No Color

# Configurações
DOMAIN="RNV.LOCAL"
REALM="RNV.LOCAL"
SHORT_DOMAIN="RNV"
HOSTNAME="adserver02"
IP="192.168.1.3"
PRIMARY_DC_IP="192.168.1.2"
PRIMARY_DC_HOSTNAME="adserver01"
DNS_FORWARDER="8.8.8.8"

# ============================================
# FUNÇÕES
# ============================================
log() {
    echo -e "${GREEN}[$(date '+%Y-%m-%d %H:%M:%S')]${NC} $1"
}

error() {
    echo -e "${RED}[ERRO]${NC} $1"
    exit 1
}

warning() {
    echo -e "${YELLOW}[AVISO]${NC} $1"
}

info() {
    echo -e "${BLUE}[INFO]${NC} $1"
}

debug() {
    echo -e "${BLUE}[DEBUG]${NC} $1"
}

check_success() {
    if [ $? -eq 0 ]; then
        echo -e "${GREEN}✓ OK${NC}"
    else
        echo -e "${RED}✗ FALHOU${NC}"
        return 1
    fi
}

# Verificar se é root
if [[ $EUID -ne 0 ]]; then
   error "Este script deve ser executado como root (sudo)"
fi

# ============================================
# CONFIGURAR LOCALE PT-BR
# ============================================
log "0. Configurando idioma Português Brasil..."

# Atualizar lista de pacotes
apt update -qq

# Instalar pacotes de idioma
apt install -y -qq language-pack-pt locales

# Gerar locale pt_BR
locale-gen pt_BR.UTF-8

# Configurar locale no sistema
update-locale LANG=pt_BR.UTF-8 LANGUAGE=pt_BR:pt LC_ALL=pt_BR.UTF-8

# Aplicar configuração
export LANG=pt_BR.UTF-8
export LANGUAGE=pt_BR:pt
export LC_ALL=pt_BR.UTF-8

echo -e "${GREEN}✓ Idioma Português Brasil configurado${NC}"
echo ""

# ============================================
# ATIVAR AUTO COMPLETE COM TAB
# ============================================
log "0. Ativando auto complete com TAB..."

# Instalar bash-completion
apt install -y -qq bash-completion

# Configurar bashrc para auto complete
if ! grep -q "bash-completion" /root/.bashrc; then
    cat >> /root/.bashrc << 'EOF'

# Ativar auto complete com TAB
if [ -f /etc/bash_completion ]; then
    . /etc/bash_completion
fi

# Ativar auto complete para samba-tool
complete -C samba-tool samba-tool
EOF
fi

# Ativar para o usuário atual
if [ -f /etc/bash_completion ]; then
    . /etc/bash_completion
fi

echo -e "${GREEN}✓ Auto complete com TAB ativado${NC}"
echo ""

# ============================================
# DETECTAR INTERFACE DE REDE
# ============================================
log "0. Detectando interface de rede..."

# Descobrir interface de rede
INTERFACE=$(ip link show | grep -E "^[0-9]+: (en|eth)" | head -1 | cut -d: -f2 | xargs)
if [ -z "$INTERFACE" ]; then
    INTERFACE=$(ip link show | grep -E "^[0-9]+: " | grep -v lo | head -1 | cut -d: -f2 | xargs)
fi

echo -e "${GREEN}✓ Interface detectada: ${INTERFACE}${NC}"
echo ""

# ============================================
# SOLICITAR SENHA DO ADMINISTRADOR
# ============================================
clear
echo -e "${BLUE}"
echo "╔═══════════════════════════════════════════════════════════════╗"
echo "║                                                               ║"
echo "║     SAMBA AD SECUNDÁRIO - INSTALAÇÃO COMPLETA                ║"
echo "║     Domínio: ${DOMAIN}                                        ║"
echo "║     Servidor: ${HOSTNAME} (Replica)                          ║"
echo "║     IP: ${IP}                                                ║"
echo "║     AD Primário: ${PRIMARY_DC_HOSTNAME} (${PRIMARY_DC_IP})   ║"
echo "║                                                               ║"
echo "╚═══════════════════════════════════════════════════════════════╝"
echo -e "${NC}"

log "=== INICIANDO CONFIGURAÇÃO DO AD SERVER SECUNDÁRIO ==="
info "Interface detectada: ${INTERFACE}"
info "IP configurado: ${IP}"
echo ""

# Solicitar senha do Administrador (mesma do domínio)
echo -e "${YELLOW}═══════════════════════════════════════════════════════════════════${NC}"
echo -e "${BLUE}Configuração da senha do Administrador do Domínio${NC}"
echo -e "${YELLOW}═══════════════════════════════════════════════════════════════════${NC}"
echo ""
echo -e "${YELLOW}⚠️  ATENÇÃO: Use a MESMA senha do administrador configurada no AdServer01${NC}"
echo ""

while true; do
    echo -e "${BLUE}Digite a senha do administrador:${NC}"
    read -s -p "> " ADMIN_PASSWORD
    echo ""
    
    echo -e "${BLUE}Confirme a senha:${NC}"
    read -s -p "> " ADMIN_PASSWORD_CONFIRM
    echo ""
    
    if [ "$ADMIN_PASSWORD" = "$ADMIN_PASSWORD_CONFIRM" ]; then
        if [ ${#ADMIN_PASSWORD} -lt 8 ]; then
            echo -e "${RED}A senha deve ter pelo menos 8 caracteres!${NC}"
            echo ""
            continue
        fi
        break
    else
        echo -e "${RED}As senhas não coincidem! Tente novamente.${NC}"
        echo ""
    fi
done

echo ""
echo -e "${GREEN}✓ Senha definida com sucesso!${NC}"
echo ""

# Perguntar se deseja continuar
read -p "Deseja continuar com a instalação? (S/n): " -n 1 -r
echo
if [[ ! $REPLY =~ ^[Ss]$ ]] && [[ ! -z "$REPLY" ]]; then
    error "Instalação cancelada pelo usuário"
fi

# ============================================
# PRÉ-VERIFICAÇÕES
# ============================================

# Verificar conectividade com o AD Primário
log "0. Verificando conectividade com o AD Primário..."

if ! ping -c 3 -W 2 ${PRIMARY_DC_IP} > /dev/null 2>&1; then
    error "Não foi possível acessar o AD Primário (${PRIMARY_DC_IP}). Verifique a rede."
fi
echo -e "${GREEN}✓ Conectividade com ${PRIMARY_DC_HOSTNAME} OK${NC}"

# Testar DNS do AD Primário
if ! host -t A ${PRIMARY_DC_HOSTNAME}.${DOMAIN,,} ${PRIMARY_DC_IP} > /dev/null 2>&1; then
    warning "Não foi possível resolver ${PRIMARY_DC_HOSTNAME}.${DOMAIN,,}"
    echo "${PRIMARY_DC_IP} ${PRIMARY_DC_HOSTNAME}.${DOMAIN,,} ${PRIMARY_DC_HOSTNAME}" >> /etc/hosts
fi
echo ""

# ============================================
# INÍCIO DA INSTALAÇÃO
# ============================================

# 1. Configurar hostname
log "1. Configurando hostname..."
hostnamectl set-hostname ${HOSTNAME}.${DOMAIN,,}
check_success

# 2. Configurar /etc/hosts
log "2. Configurando /etc/hosts..."
if [ -f /etc/hosts ]; then
    cp /etc/hosts /etc/hosts.bak.$(date +%Y%m%d_%H%M%S) 2>/dev/null || true
fi

cat > /etc/hosts << EOF
127.0.0.1 localhost
127.0.1.1 ${HOSTNAME}.${DOMAIN,,} ${HOSTNAME}
${IP} ${HOSTNAME}.${DOMAIN,,} ${HOSTNAME}
${PRIMARY_DC_IP} ${PRIMARY_DC_HOSTNAME}.${DOMAIN,,} ${PRIMARY_DC_HOSTNAME}
EOF
check_success

# 3. Configurar NTP
log "3. Instalando e configurando NTP..."
apt update -qq
apt install chrony -y -qq
timedatectl set-timezone America/Sao_Paulo

# Configurar chrony para sincronizar com o AD Primário
cat > /etc/chrony/chrony.conf << EOF
pool ${PRIMARY_DC_IP} iburst
driftfile /var/lib/chrony/chrony.drift
keyfile /etc/chrony/chrony.keys
logdir /var/log/chrony
EOF

systemctl enable --now chrony -qq
check_success

# 4. Instalar pacotes do Samba
log "4. Instalando pacotes do Samba AD..."
DEBIAN_FRONTEND=noninteractive apt install -y -qq \
    acl attr samba samba-dsdb-modules \
    samba-vfs-modules winbind libpam-winbind \
    libnss-winbind kinit krb5-user dnsutils \
    bind9utils ldap-utils \
    bash-completion \
    language-pack-pt \
    locales
check_success

# 5. Parar serviços conflitantes
log "5. Parando serviços conflitantes..."
systemctl stop smbd nmbd winbind 2>/dev/null || true
systemctl disable smbd nmbd winbind 2>/dev/null || true
check_success

# 6. Limpar configurações antigas
log "6. Limpando configurações antigas..."
rm -f /etc/samba/smb.conf
rm -rf /var/lib/samba/private 2>/dev/null || true
rm -rf /var/lib/samba/sysvol 2>/dev/null || true
rm -f /etc/krb5.conf
check_success

# 7. Configurar resolv.conf (usar AD Primário primeiro)
log "7. Configurando resolução DNS..."

if lsattr /etc/resolv.conf 2>/dev/null | grep -q "i"; then
    info "Removendo atributo imutável do /etc/resolv.conf..."
    chattr -i /etc/resolv.conf 2>/dev/null || true
fi

if [ -f /etc/resolv.conf ]; then
    cp /etc/resolv.conf /etc/resolv.conf.bak.$(date +%Y%m%d_%H%M%S) 2>/dev/null || true
fi

cat > /etc/resolv.conf << EOF
nameserver ${PRIMARY_DC_IP}
nameserver 127.0.0.1
search ${DOMAIN,,}
domain ${DOMAIN,,}
EOF

info "Arquivo /etc/resolv.conf atualizado"
check_success

# 8. Configurar Kerberos
log "8. Configurando Kerberos..."

cat > /etc/krb5.conf << EOF
[libdefaults]
    default_realm = ${REALM}
    dns_lookup_realm = false
    dns_lookup_kdc = true
    ticket_lifetime = 24h
    renew_lifetime = 7d
    forwardable = true

[realms]
    ${REALM} = {
        kdc = ${PRIMARY_DC_HOSTNAME}.${DOMAIN,,}
        admin_server = ${PRIMARY_DC_HOSTNAME}.${DOMAIN,,}
        default_domain = ${DOMAIN,,}
    }

[domain_realm]
    .${DOMAIN,,} = ${REALM}
    ${DOMAIN,,} = ${REALM}

[logging]
    kdc = FILE:/var/log/krb5kdc.log
    admin_server = FILE:/var/log/kadmin.log
    default = FILE:/var/log/krb5lib.log
EOF

check_success

# 9. Testar Kerberos antes de provisionar
log "9. Testando autenticação Kerberos no AD Primário..."

echo "═══════════════════════════════════════════════════════════════════"
debug "Testando autenticação com o AD Primário..."
debug "Domínio: ${REALM}"
debug "Usuário: administrator@${REALM}"
echo "═══════════════════════════════════════════════════════════════════"

if echo "${ADMIN_PASSWORD}" | kinit administrator@${REALM} > /dev/null 2>&1; then
    echo -e "${GREEN}✅ Kerberos funcionando com o AD Primário!${NC}"
    klist
else
    echo -e "${RED}❌ Kerberos falhou - Não é possível continuar${NC}"
    error "Não foi possível autenticar no AD Primário. Verifique a senha."
fi

echo "═══════════════════════════════════════════════════════════════════"
echo ""

# ============================================
# 10. JOIN NO DOMÍNIO - CORRIGIDO
# ============================================
log "10. Provisionando como controlador secundário (replica)..."
info "Isso pode levar alguns minutos..."

echo ""
echo -e "${YELLOW}═══════════════════════════════════════════════════════════════════${NC}"
echo -e "${BLUE}Executando join no domínio...${NC}"
echo -e "${YELLOW}═══════════════════════════════════════════════════════════════════${NC}"
echo ""

# CORREÇÃO: O join NÃO aceita --host-ip nem --adminpass
# Usamos apenas as opções válidas
if echo "${ADMIN_PASSWORD}" | samba-tool domain join ${DOMAIN} DC \
    -U administrator \
    --realm=${REALM} \
    --dns-backend=SAMBA_INTERNAL \
    2>&1 | tee /tmp/join.log; then
    
    # Verificar se o join foi bem sucedido
    if grep -q "Joined domain" /tmp/join.log || grep -q "successfully" /tmp/join.log; then
        echo -e "${GREEN}✓ OK - Servidor secundário provisionado com sucesso!${NC}"
    else
        echo -e "${RED}✗ O join pode ter falhado.${NC}"
        echo ""
        echo -e "${YELLOW}═══════════════════════════════════════════════════════════════════${NC}"
        echo -e "${RED}ERRO DETALHADO:${NC}"
        echo -e "${YELLOW}═══════════════════════════════════════════════════════════════════${NC}"
        cat /tmp/join.log
        echo ""
        error "Falha no join. Verifique o log /tmp/join.log"
    fi
else
    echo -e "${RED}✗ FALHOU${NC}"
    echo ""
    echo -e "${YELLOW}═══════════════════════════════════════════════════════════════════${NC}"
    echo -e "${RED}ERRO DETALHADO:${NC}"
    echo -e "${YELLOW}═══════════════════════════════════════════════════════════════════${NC}"
    cat /tmp/join.log
    echo ""
    error "Falha ao provisionar servidor secundário. Verifique o log acima."
fi

echo ""

# 11. Configurar smb.conf para secundário
log "11. Configurando smb.conf..."

if [ -f "/var/lib/samba/private/smb.conf" ]; then
    cp /var/lib/samba/private/smb.conf /etc/samba/smb.conf
    info "Arquivo copiado do private"
else
    info "Criando smb.conf manualmente..."
    cat > /etc/samba/smb.conf << EOF
[global]
    netbios name = ${HOSTNAME^^}
    realm = ${REALM}
    server role = active directory domain controller
    workgroup = ${SHORT_DOMAIN}
    dns forwarder = ${DNS_FORWARDER}
    interfaces = lo ${INTERFACE}
    bind interfaces only = yes
    log level = 2
    max log size = 1000
    debug timestamp = yes
    server services = s3fs, rpc, nbt, wrepl, ldap, cldap, kdc, drepl, winbind, ntp_signd, kcc, dnsupdate, dns
EOF
fi

# Garantir configurações corretas
if ! grep -q "interfaces.*${INTERFACE}" /etc/samba/smb.conf; then
    sed -i "s/interfaces.*=.*/interfaces = lo ${INTERFACE}/g" /etc/samba/smb.conf
fi

if ! grep -q "dns forwarder" /etc/samba/smb.conf; then
    echo "    dns forwarder = ${DNS_FORWARDER}" >> /etc/samba/smb.conf
fi

check_success

# 12. Iniciar serviços
log "12. Iniciando serviços do Samba AD..."
systemctl unmask samba-ad-dc 2>/dev/null || true
systemctl enable samba-ad-dc 2>/dev/null || true
systemctl restart samba-ad-dc

# Verificar se o serviço iniciou
sleep 5
if systemctl is-active --quiet samba-ad-dc; then
    echo -e "${GREEN}✓ Serviço iniciado com sucesso${NC}"
else
    warning "Serviço não iniciou automaticamente. Tentando novamente..."
    systemctl restart samba-ad-dc
    sleep 5
    if systemctl is-active --quiet samba-ad-dc; then
        echo -e "${GREEN}✓ Serviço iniciado na segunda tentativa${NC}"
    else
        warning "Serviço ainda não iniciou. Verifique manualmente."
        systemctl status samba-ad-dc --no-pager
    fi
fi

# 13. Aguardar serviços iniciarem
log "13. Aguardando serviços iniciarem (30 segundos)..."
sleep 30

# 14. Ajustar resolv.conf para usar o próprio servidor
log "14. Ajustando resolução DNS para usar o servidor local..."

cat > /etc/resolv.conf << EOF
nameserver 127.0.0.1
nameserver ${PRIMARY_DC_IP}
search ${DOMAIN,,}
domain ${DOMAIN,,}
EOF
check_success

# 15. Verificar DNS
log "15. Verificando DNS..."
info "Testando resolução DNS..."

sleep 5
if host -t SRV _ldap._tcp.${DOMAIN,,} > /dev/null 2>&1; then
    echo -e "${GREEN}✓ DNS SRV OK${NC}"
else
    warning "DNS SRV não resolvido corretamente - pode levar mais tempo"
    sleep 10
    host -t SRV _ldap._tcp.${DOMAIN,,} || warning "DNS SRV ainda não disponível"
fi

if host -t A ${HOSTNAME}.${DOMAIN,,} > /dev/null 2>&1; then
    echo -e "${GREEN}✓ Hostname OK${NC}"
else
    warning "Hostname não resolvido - verificando..."
    sleep 5
    host -t A ${HOSTNAME}.${DOMAIN,,} || warning "Hostname ainda não disponível"
fi

# 16. Testar replicação
log "16. Testando replicação..."

# Testar DNS do domínio
if host -t SRV _ldap._tcp.${DOMAIN,,} | grep -q "${HOSTNAME}"; then
    echo -e "${GREEN}✓ ${HOSTNAME} encontrado nos registros DNS${NC}"
else
    warning "${HOSTNAME} não encontrado nos registros DNS"
fi

# Testar usuários replicados
if samba-tool user list 2>/dev/null | grep -q "administrator"; then
    echo -e "${GREEN}✓ Usuários replicados com sucesso${NC}"
else
    warning "Não foi possível listar usuários - pode levar mais tempo"
fi

# 17. Verificar status do serviço
log "17. Verificando status do serviço..."
systemctl status samba-ad-dc --no-pager | grep -E "Active:|Loaded:" || warning "Serviço não está ativo"

# 18. Salvar informações
log "18. Salvando informações de configuração..."
cat > /root/ad_info.txt << EOF
═══════════════════════════════════════════════════════════════════
                    INFORMAÇÕES DO DOMÍNIO
═══════════════════════════════════════════════════════════════════

DOMÍNIO: ${DOMAIN}
REALM: ${REALM}
IP: ${IP}
SERVIDOR: ${HOSTNAME} (SECUNDÁRIO)
INTERFACE: ${INTERFACE}
AD PRIMÁRIO: ${PRIMARY_DC_HOSTNAME} (${PRIMARY_DC_IP})
DATA CRIAÇÃO: $(date)

───────────────────────────────────────────────────────────────────
USUÁRIOS E SENHAS
───────────────────────────────────────────────────────────────────
Administrador: administrator@${REALM}
Senha: ${ADMIN_PASSWORD}

⚠️  IMPORTANTE: Guarde estas senhas em local seguro!

───────────────────────────────────────────────────────────────────
RESULTADO DO TESTE KERBEROS
───────────────────────────────────────────────────────────────────
$(klist 2>&1 || echo "Sem ticket ativo")

───────────────────────────────────────────────────────────────────
COMANDOS ÚTEIS
───────────────────────────────────────────────────────────────────
# Testar Kerberos
echo '${ADMIN_PASSWORD}' | kinit administrator@${REALM}

# Verificar ticket
klist

# Listar usuários
samba-tool user list

# Listar grupos
samba-tool group list

# Testar DNS
host -t SRV _ldap._tcp.${DOMAIN,,}
host -t A ${HOSTNAME}.${DOMAIN,,}

# Verificar replicação
samba-tool drs showrepl

───────────────────────────────────────────────────────────────────
LOGS
───────────────────────────────────────────────────────────────────
Arquivo de log: /tmp/join.log
Log do Samba: /var/log/samba/log.samba
EOF

check_success

# 19. Mostrar resumo final
clear
echo -e "${GREEN}"
echo "═══════════════════════════════════════════════════════════════════"
echo "                                                                   "
echo "         ✅ AD SECUNDÁRIO CONFIGURADO COM SUCESSO!                 "
echo "                                                                   "
echo "═══════════════════════════════════════════════════════════════════"
echo -e "${NC}"
echo ""
echo -e "${BLUE}📋 RESUMO DA INSTALAÇÃO:${NC}"
echo ""
echo -e "  ${GREEN}✓${NC} Idioma Português Brasil configurado"
echo -e "  ${GREEN}✓${NC} Auto complete com TAB ativado"
echo -e "  ${GREEN}✓${NC} Hostname configurado: ${HOSTNAME}.${DOMAIN,,}"
echo -e "  ${GREEN}✓${NC} Interface detectada: ${INTERFACE} (${IP})"
echo -e "  ${GREEN}✓${NC} /etc/hosts configurado"
echo -e "  ${GREEN}✓${NC} Conectividade com AD Primário OK"
echo -e "  ${GREEN}✓${NC} Kerberos funcionando"
echo -e "  ${GREEN}✓${NC} Join no domínio como DC secundário"
echo -e "  ${GREEN}✓${NC} Sincronização com AD Primário"
echo -e "  ${GREEN}✓${NC} Serviços iniciados"
echo -e "  ${GREEN}✓${NC} DNS configurado"
echo ""
echo -e "${YELLOW}📌 INFORMAÇÕES DE ACESSO:${NC}"
echo ""
echo -e "  Domínio: ${BLUE}${DOMAIN}${NC}"
echo -e "  Usuário Admin: ${BLUE}administrator@${REALM}${NC}"
echo -e "  Senha Admin: ${BLUE}${ADMIN_PASSWORD}${NC}"
echo -e "  IP do Servidor: ${BLUE}${IP}${NC}"
echo -e "  Interface: ${BLUE}${INTERFACE}${NC}"
echo -e "  AD Primário: ${BLUE}${PRIMARY_DC_HOSTNAME} (${PRIMARY_DC_IP})${NC}"
echo ""
echo -e "${YELLOW}📁 ARQUIVOS SALVOS:${NC}"
echo ""
echo -e "  /root/ad_info.txt - Informações completas"
echo -e "  /tmp/join.log - Log do join"
echo ""
echo -e "${YELLOW}🔧 TESTES MANUAIS:${NC}"
echo ""
echo -e "  Para testar o Kerberos manualmente, execute:"
echo -e "  ${BLUE}echo '${ADMIN_PASSWORD}' | kinit administrator@${REALM}${NC}"
echo -e "  ${BLUE}klist${NC}"
echo ""
echo -e "  Para verificar a replicação:"
echo -e "  ${BLUE}samba-tool drs showrepl${NC}"
echo ""
echo -e "${YELLOW}💡 DICAS:${NC}"
echo -e "  ✅ Use TAB para auto completar comandos"
echo -e "  ✅ O sistema está em Português Brasil"
echo -e "  ✅ Este servidor é uma réplica do AD Primário"
echo -e "  ✅ Em caso de falha do AD Primário, este servidor pode assumir"
echo ""
echo -e "${GREEN}═══════════════════════════════════════════════════════════════════${NC}"
echo -e "${GREEN}                    FIM DA INSTALAÇÃO                              ${NC}"
echo -e "${GREEN}═══════════════════════════════════════════════════════════════════${NC}"
echo ""

# Perguntar se deseja reiniciar
read -p "Deseja reiniciar o servidor agora? (s/N): " -n 1 -r
echo
if [[ $REPLY =~ ^[Ss]$ ]]; then
    log "Reiniciando servidor em 5 segundos..."
    sleep 5
    reboot
else
    log "Lembre-se de reiniciar o servidor depois para aplicar todas as configurações."
fi
